procedure TTCliente.BtnEditarClick(Sender: TObject);

end;


procedure TTCliente.BtnExcluirClick(Sender: TObject);

      QCliente.Delete;


    end;
  end;
end;

procedure TTCliente.ChamaTela;
begin
  TCliente := TTCliente.Create(Application);
  with TCliente do
  begin
    ShowModal;
    FreeAndNil(TCliente);
  end;
end;




procedure TTCliente.edPesquisaKeyPress(Sender: TObject; var Key: Char);
var
  vCodigo: Integer;
begin
  if (Key = #13) then
  begin
    with QCliente do
    begin
      Close;
      SQL.Clear;
      SQL.Add('SELECT COD_CLIENT, NOME,CAST(ATIVO AS INTEGER) AS ATIVO');
      SQL.Add('FROM CLIENTE');
      if (edPesquisa.Text = '*') then
//      begin
//        SQL.Add('ORDER BY NOME');
//      end
      else if TryStrToInt(edPesquisa.Text, vCodigo) then
      begin
        SQL.Add('WHERE COD_CLIENT = ' + edPesquisa.Text);
      end
      else if (edPesquisa.Text <> '') then
      begin
        SQL.Add('WHERE NOME LIKE ''%' + edPesquisa.Text + '%''');
      end
       else if (edPesquisa.Text <> '') then
       BEGIN
           SQL.Add('WHERE ATIVO = 0 ' + edPesquisa.Text );
       END;


      Open;
    end;
  end;
end;



procedure TTCliente.LabNumeroRegistroClick(Sender: TObject);
begin
with Qcliente do
begin
  close;
  SQL.Clear;
    SQL.Add('SELECT COUNT(*) AS TOTAL FROM CLIENTE');
    open;
    labNumeroRegistro.Caption := FieldByName('TOTAL').AsString;
     close;
     open;
end;

end;

procedure TTCliente.RadioGroupClick(Sender: TObject);
begin
  with QCliente do
  begin
    Close;
    SQL.Clear;
    SQL.Add('SELECT COD_CLIENT, NOME, CAST(ATIVO AS INTEGER) AS ATIVO');
    SQL.Add('FROM CLIENTE');
    SQL.Add('WHERE 1=1');
    if Trim(edPesquisa.Text) <> '' then
    begin
      SQL.Add('AND NOME LIKE ' + QuotedStr('%' + edPesquisa.Text + '%'));
    end;
    case RadioGroup.ItemIndex of
      0: SQL.Add('AND ATIVO = 1');
      1: SQL.Add('AND ATIVO = 0');
    end;
    Open;
  end;
end;
