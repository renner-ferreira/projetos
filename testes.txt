   // Filtro do RadioGroup

procedure TForm1.btnExcluirClick(Sender: TObject);
begin
  // 1. Verifica se o DataSet está ativo e tem registros
  if not cdsDados.Active or cdsDados.IsEmpty then
  begin
    ShowMessage('Não há registros para excluir.');
    Exit;
  end;

  // 2. Confirmação do usuário
  if MessageDlg('Deseja realmente excluir o registro selecionado?',
    mtConfirmation, [mbYes, mbNo], 0) = mrYes then
  begin
    try
      // 3. Executa a exclusão
      cdsDados.Delete;
      
      // 4. Aplica as mudanças (se usar ClientDataSet + Provider)
      if cdsDados.ChangeCount > 0 then
        cdsDados.ApplyUpdates(0);
        
    except
      on E: Exception do
        ShowMessage('Erro ao excluir: ' + E.Message);
    end;
  end;
end;







chama tela 
  type
  TClienteC = class(TForm)

  public

    class procedure ChamaTela(ADataSet: TDataSet);
  end;

class procedure TClienteC.ChamaTela(ADataSet: TDataSet);
begin
  if ADataSet.IsEmpty then Exit;


  Application.CreateForm(TClienteC, ClienteC);
  try

    ADataSet.Edit; 
    

    if ClienteC.ShowModal = mrOk then
    begin

      if ADataSet.State in [dsEdit, dsInsert] then
        ADataSet.Post;
        
      if ADataSet is TClientDataSet then
        TClientDataSet(ADataSet).ApplyUpdates(0);
    end
    else
    begin

      ADataSet.Cancel;
    end;
  finally
    ClienteC.Free;
  end;
end;
